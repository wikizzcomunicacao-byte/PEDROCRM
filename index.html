async function handleLogin(e) { e.preventDefault(); const email = document.getElementById('login-email').value.trim(); const password = document.getElementById('login-password').value; const btn = document.getElementById('btn-submit-login'); const errorMsg = document.getElementById('login-error-msg'); errorMsg.classList.add('hidden'); if (email.toLowerCase() !== EMAIL_AUTORIZADO.toLowerCase()) { showLoginError('Acesso negado: Este e-mail não tem permissão para aceder a este sistema.'); return; } btn.disabled = true; btn.innerHTML = ` Verificando...`; try { console.log("A tentar autenticar com:", email); const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password }); if (error) { console.error("Erro retornado pelo Supabase:", error); throw error; } console.log("Login efetuado com sucesso!", data); currentUser = data.user; showAppScreen(); await syncData(); } catch (err) { console.error("Catch capturou o erro:", err); showLoginError(err.message || 'Erro ao efetuar login. Verifique as credenciais.'); } finally { btn.disabled = false; btn.innerHTML = ` Entrar no Sistema`; } }

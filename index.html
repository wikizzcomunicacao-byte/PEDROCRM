Script SQL para o Supabase

Copie este código e cole no SQL Editor do seu projeto Supabase

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE IF NOT EXISTS leads (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nome VARCHAR(100) NOT NULL,
    whatsapp VARCHAR(20) NOT NULL,
    email VARCHAR(100),
    tipo_bem VARCHAR(30) DEFAULT 'imovel',
    credito_pretendido DECIMAL(12,2) DEFAULT 0,
    etapa_funil VARCHAR(30) DEFAULT 'novo_lead',
    temperatura VARCHAR(20) DEFAULT 'morno',
    observacoes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS cotas (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    cliente_nome VARCHAR(100) NOT NULL,
    administradora VARCHAR(50) NOT NULL,
    grupo VARCHAR(20) NOT NULL,
    cota VARCHAR(20) NOT NULL,
    valor_credito DECIMAL(12,2) NOT NULL,
    valor_parcela DECIMAL(10,2) NOT NULL,
    dia_assembleia INT DEFAULT 15,
    status_cota VARCHAR(30) DEFAULT 'ativa',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

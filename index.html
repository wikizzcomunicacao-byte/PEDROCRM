Script SQL para Criação das Tabelas no Supabase

Para que a sincronização funcione perfeitamente, copie o código SQL abaixo e rode no SQL Editor dentro do seu painel do Supabase: 

-- 1. Tabela de Leads / Funil de Vendas
CREATE TABLE IF NOT EXISTS leads (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nome VARCHAR(100) NOT NULL,
    whatsapp VARCHAR(20) NOT NULL,
    tipo VARCHAR(30) DEFAULT 'Imóvel',
    credito DECIMAL(12,2) DEFAULT 0,
    etapa VARCHAR(30) DEFAULT 'novo_lead',
    obs TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Tabela de Cotas
CREATE TABLE IF NOT EXISTS cotas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    lead_id UUID REFERENCES leads(id) ON DELETE SET NULL,
    administradora VARCHAR(50) NOT NULL,
    grupo VARCHAR(20) NOT NULL,
    cota VARCHAR(20) NOT NULL,
    credito DECIMAL(12,2) NOT NULL,
    parcela DECIMAL(10,2) NOT NULL,
    vencimento INT CHECK (vencimento BETWEEN 1 AND 31),
    assembleia INT CHECK (assembleia BETWEEN 1 AND 31),
    lance VARCHAR(50) DEFAULT 'Livre',
    status VARCHAR(30) DEFAULT 'ativa',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Tabela de Lances
CREATE TABLE IF NOT EXISTS lances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    cota_id UUID REFERENCES cotas(id) ON DELETE CASCADE,
    mes_ano VARCHAR(7) NOT NULL,
    tipo_lance VARCHAR(30) NOT NULL,
    percentual DECIMAL(5,2) NOT NULL,
    valor DECIMAL(12,2),
    contemplado BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Desabilitar RLS para acesso anonimo simples
ALTER TABLE leads DISABLE ROW LEVEL SECURITY;
ALTER TABLE cotas DISABLE ROW LEVEL SECURITY;
ALTER TABLE lances DISABLE ROW LEVEL SECURITY;
